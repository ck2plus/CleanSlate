# IDs JD.4000 - 4030

namespace = JD

character_event = { # Yearly event for Raiding China
	id = JD.4000

	hide_window = yes
	is_triggered_only = yes # on_yearly_pulse

	has_dlc = "Jade Dragon"

	trigger = {
		has_character_modifier = china_raid_active

		offmap_china = {
			NOT = {
				governor = {
					war_with = ROOT
				}
			}
		}
	}

	immediate = {
		if = { # Remove raiding if no longer allowed to raid
			limit = {
				OR = {
					independent = no
					is_allowed_to_loot = no
					is_landed = no
				}
			}

			remove_character_modifier = china_raid_active

			add_character_modifier = {
				name = china_raid_grace_cd
				years = 5
				hidden = yes
			}

			character_event = { id = JD.4001 }
		}
		else = {
			random_list = { # All raids give Gold + Prestige and Population for Nomads
				400 = { # Only Gold + Prestige
					character_event = { id = JD.4020 }
				}

				30 = { # Big amount of gold + prestige
					character_event = { id = JD.4021 }
				}

				30 = { # Captured siege engineer
					character_event = { id = JD.4022 }
				}

				30 = { # Attractive opposite sex character
					trigger = { is_adult = yes } #Since this is to take a concubine
					character_event = { id = JD.4023 }
				}

				30 = { # Bad outcome
					character_event = { id = JD.4024 }
				}

				15 = { # Minor artifact
					character_event = { id = JD.4025 }
				}

				5 = { # An enormous amount of gold + prestige
					character_event = { id = JD.4026 }
				}

				5 = { # A panda
					trigger = {
						NOT = { has_character_modifier = panda_pet }
					}

					character_event = { id = JD.4027 }
				}
			}
		}
	}
}

character_event = { #Siege Engineer
	id = JD.4022
	desc = EVTDESC_JD_4022
	picture = GFX_evt_china_unrest
	border = GFX_event_normal_frame_war
	
	is_triggered_only = yes
	
	has_dlc = "Jade Dragon"
	
	immediate = {
		create_character = {
			dynasty = actually_culture
			religion = taoist
			culture = han
			female = no
			health = 6
			random_traits = yes

			attributes = {
				diplomacy = 2
				learning = 2
				stewardship = 2
				intrigue = 2
				martial = 13
			}
		}

		new_character = {
			set_character_flag = originated_from_chinese_court
			set_character_flag = no_court_invites
			set_character_flag = taught_chinese_strategy
			set_character_flag = is_chinese_siege_engineer
			set_character_flag = ignore_refund

			remove_trait = weak
			remove_trait = slow
			remove_trait = imbecile
			remove_trait = dull

			remove_lifestyle_trait_effect = yes

			add_trait = sapper

			if = {
				limit = {
					offmap_china = { has_status = china_golden_age }
				}

				change_martial = 5
			}

			random_list = { # Flavor Randomization
				15 = { change_martial = 2 }
				10 = { add_trait = brawny }
				10 = { add_trait = duelist }
				10 = { add_trait = architect }
				 5 = { add_trait = strategist }
				 5 = { add_trait = shrewd }
				 5 = { change_martial = 4 }

				 5 = {
					if = {
						limit = { has_dlc = "Reapers" }
						add_trait = one_eyed
					}
				}
			}

			save_event_target_as = captured_engineer
		}
	}
	
	option = { #Bring them to court
		name = EVTOPTA_JD_4022
	}
	
	option = { #Make them a commander
		name = EVTOPTB_JD_4022
		
		custom_tooltip = {
			text = fire_commander_tt

			event_target:captured_engineer = {
				if = {
					limit = { can_hold_title = title_commander }

					if = {
						limit = {
							ROOT = { can_grant_title = title_commander }
						}

						give_minor_title = title_commander
					}
					else = {
						ROOT = {
							random_courtier_or_vassal = {
								limit = {
									has_minor_title = title_commander

									NOR = {
										has_character_flag = originated_from_chinese_court
										has_character_flag = taught_chinese_strategy
									}
								}

								preferred_limit = {
									lower_real_tier_than = COUNT
									martial < 10
								}

								preferred_limit = {
									lower_real_tier_than = COUNT
								}

								remove_title = title_commander
							}
						}

						give_minor_title = title_commander
					}
				}
			}
		}
	}
	
	option = { #We have no need of him
		name = EVTOPTC_JD_4022
		event_target:captured_engineer = {
			death = { death_reason = death_missing }
		}
	}
	
	after = {
		prestige = 75

		if = {
			limit = { is_nomadic = yes }
			scaled_wealth = { value = 0.6 min = 70 max = 175 }
			population = 150
		}
		else = {
			scaled_wealth = { value = 0.6 min = 30 max = 125 }
		}

		reverse_opinion = {
			name = opinion_abducted
			who = event_target:captured_engineer
			years = 10
		}
	}
}

character_event = { #Attractive character
	id = JD.4023
	desc = EVTDESC_JD_4023
	picture = GFX_evt_china_unrest
	border = GFX_event_normal_frame_war
	
	is_triggered_only = yes
	
	has_dlc = "Jade Dragon"
	
	immediate = {
		if = {
			limit = { is_female = yes }

			create_character = {
				age = 20
				dynasty = actually_culture
				religion = taoist
				culture = han
				female = no
				health = 6
				random_traits = yes
			}
		}
		else = {
			create_character = {
				age = 20
				dynasty = actually_culture
				religion = taoist
				culture = han
				female = yes
				health = 6
				random_traits = yes
			}
		}

		new_character = {
			remove_all_negative_congenital_traits_effect = yes
			add_trait = attractive

			random = {
				chance = 35
				add_trait = strong
			}

			random = {
				chance = 35
				add_trait = quick
			}

			save_event_target_as = captured_character
		}
	}
	
	option = { #Bring them to my court
		name = EVTOPTA_JD_4023
	}
	
	option = { #Make them a concubine
		name = EVTOPTB_JD_4023
		
		trigger = {
			is_female = no
			has_concubinage = yes
			num_of_consorts < 3
		}

		add_consort = event_target:captured_character
	}
	
	option = { #Make them a consort
		name = EVTOPTB_JD_4023
		
		trigger = {
			is_female = yes
			has_concubinage = yes #This automatically filters out polygamists and all doctrines that include it, hurray!
			num_of_consorts < 3
			OR = { #Follows a religion that allows women to take consorts
				religion_openly_west_african_or_reformed_trigger = yes
				religion_openly_bon_or_reformed_trigger = yes
				religion = { has_religion_feature = religion_matriarchal }
				religion = { has_religion_feature = religion_equal }
			}
			NOT = { #And hasn't removed that capacity
				religion = { has_religion_feature = religion_patriarchal }
			}
			
		}
		
		add_consort = event_target:captured_character
	}
	
	option = { #I have no use for them
		name = EVTOPTC_JD_4023
		event_target:captured_engineer = {
			death = { death_reason = death_missing }
		}
	}
	
	after = {
		prestige = 75

		if = {
			limit = { is_nomadic = yes }
			scaled_wealth = { value = 0.6 min = 70 max = 175 }
			population = 150
		}
		else = {
			scaled_wealth = { value = 0.6 min = 30 max = 125 }
		}

		reverse_opinion = {
			name = opinion_abducted
			who = event_target:captured_character
			years = 10
		}
	}
}
