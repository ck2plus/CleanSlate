buddhist_holy_war = {
	name = CB_NAME_RELIGIOUS_MINOR
	war_name = WAR_NAME_RELIGIOUS
	sprite = 9
	truce_days = 1825
	sort_priority = 800

	hostile_against_others = yes
	is_permanent = yes
	check_all_titles = yes # if permanent, setting this to true will check against all of someones titles, including vassal held titles
	can_ask_to_join_war = no
	allowed_to_target_tributaries = no

	can_use = {
		# Attacker
		ROOT = {
			religion = buddhist
			NOT = { same_realm = FROM }

			trigger_if = {
				limit = { religion_group = FROM }

				OR = {
					FROM = { religion = taoist }

					# Can always attack known devil worshippers of the same religion
					AND = {
						FROM = { has_character_modifier = known_satanist }
						NOT = { has_character_modifier = known_satanist }
					}
				}
			}

			top_liege = {
				religion_group = ROOT

				OR = {
					NOT = { religion = FROM }
					FROM = { religion = taoist }

					# Can always attack known devil worshippers of the same religion
					trigger_if = {
						limit = { religion_group = FROM }

						FROM = { has_character_modifier = known_satanist }
						NOT = { has_character_modifier = known_satanist }

						ROOT = {
							NOT = { has_character_modifier = known_satanist }
						}
					}
				}
			}

			is_nomadic = no
			mercenary = no
		}
	}

	can_use_title = {
		real_tier = COUNT

		# The attacker needs a border or be at the most two sea zones away
		location = {
			any_neighbor_province = {
				OR = {
					trigger_if = {
						limit = { has_owner = yes }

						owner = {
							OR = {
								character = ROOT
								is_vassal_or_below_of = ROOT
							}
						}
					}
					trigger_else = {
						is_land = no # first sea zone

						any_neighbor_province = {
							OR = {
								trigger_if = {
									limit = { has_owner = yes }

									owner = {
										OR = {
											character = ROOT
											is_vassal_or_below_of = ROOT
										}
									}
								}
								trigger_else = {
									is_land = no # second sea zone

									any_neighbor_province = {
										owner = {
											OR = {
												character = ROOT
												is_vassal_or_below_of = ROOT
											}
										}
									}
								}
							}
						}
					}
				}
			}
		}
	}

	is_valid = {
		ROOT = {
			religion = buddhist
			NOT = { same_realm = FROM }

			trigger_if = {
				limit = { religion_group = FROM }

				OR = {
					FROM = { religion = taoist }

					# Can always attack known devil worshippers of the same religion
					AND = {
						FROM = { has_character_modifier = known_satanist }
						NOT = { has_character_modifier = known_satanist }
					}
				}
			}

			top_liege = {
				OR = {
					NOT = { religion = FROM }
					FROM = { religion = taoist }

					AND = { # Can always attack known devil worshippers of the same religion
						religion_group = FROM
						FROM = { has_character_modifier = known_satanist }
						NOT = { has_character_modifier = known_satanist }

						ROOT = {
							NOT = { has_character_modifier = known_satanist }
						}
					}
				}
			}
		}
	}

	on_add = {
		anger_china_effect = yes
	}

	on_success = {
		thirdparty_title_scope = {
			show_scope_change = no

			if = { # Vassalise one county counts
				limit = {
					holder_scope = {
						real_tier = COUNT
						num_of_count_titles == 1
						lower_real_tier_than = attacker
						is_patrician = no
					}
				}

				holder_scope = {
					show_scope_change = no
					set_defacto_liege = attacker
				}
			}

			else = { # Usurp title in all other circumstances
				usurp_title_plus_barony_if_unlanded = {
					target = attacker
					type = holy_war
				}

				any_de_jure_vassal_title = { # take all baronies under the one we're fighting for
					limit = {
						NOT = { de_facto_liege = PREV }

						holder_scope = {
							OR = {
								character = defender
								is_vassal_or_below_of = defender
							}
						}
					}

					usurp_title_plus_barony_if_unlanded = {
						target = attacker
						type = holy_war
					}
				}
			}
		}

		any_attacker = {
			show_scope_change = no

			if = {
				limit = { character = attacker }
				participation_scaled_prestige = 100
			}
			else = {
				hidden_effect = { participation_scaled_prestige = 100 }
			}

			hidden_effect = {
				if = {
					limit = {
						has_ambition = obj_strengthen_religion
						NOT = { religion = defender }
					}

					change_variable = {
						which = strengthen_religion
						value = 1
					}
				}
			}
		}
	}

	on_fail_title = {
		attacker = {
			show_scope_change = no
			prestige = -100
		}

		any_defender = {
			show_scope_change = no

			if = {
				limit = { character = defender }
				participation_scaled_prestige = 50
			}
			else = {
				hidden_effect = { participation_scaled_prestige = 50 }
			}
		}
	}

	on_reverse_demand = {
		attacker = {
			show_scope_change = no

			prestige = -200

			transfer_scaled_wealth = {
				to = defender
				value = 4.0
			}
		}

		any_defender = {
			show_scope_change = no

			if = {
				limit = { character = defender }
				participation_scaled_prestige = 100
			}
			else = {
				hidden_effect = {participation_scaled_prestige = 100 }
			}

			hidden_effect = {
				if = {
					limit = {
						has_ambition = obj_strengthen_religion
						NOT = { religion = attacker }
					}

					change_variable = {
						which = strengthen_religion
						value = 1
					}
				}
			}
		}
	}

	attacker_ai_victory_worth = {
		factor = -1 # always accept
	}

	attacker_ai_defeat_worth = {
		factor = 100
	}

	defender_ai_victory_worth = {
		factor = -1 # always accept
	}

	defender_ai_defeat_worth = {
		factor = 100
	}

	ai_will_do = {
		factor = 1

		trigger = {
			suitable_holdings_trigger = yes
		}
	}
}

indian_subjugation = {
	name = CB_NAME_PAGAN_SUBJUGATION
	war_name = WAR_NAME_PAGAN_SUBJUGATION
	sprite = 8
	truce_days = 365
	sort_priority = 1000

	hostile_against_others = yes
	is_permanent = yes
	can_ask_to_join_war = yes
	check_de_jure_tier = KING # this scans all de jure kingdoms for the counties which are held by or vassals(or below) of selected character. Only valid if is_permanent = yes
	allowed_to_target_tributaries = no

	can_use_gui = {
		# Attacker
		ROOT = {
			piety >= 500
		}
	}

	can_use = {
		# Attacker
		ROOT = {
			has_dharmic_religion_trigger = yes
			culture_group = FROM
			NOT = { same_realm = FROM }
			has_horde_culture = no # Should use tribal invasion instead
			is_nomadic = no
			mercenary = no
		}

		# Defender
		FROM = {
			has_dharmic_religion_trigger = yes

			NOR = {
				is_offmap_governor = yes

				any_war = {
					defender = { character = PREVPREV }
					using_cb = indian_subjugation
				}
			}
		}
	}

	can_use_title = {
		# Attacker
		ROOT = {
			trigger_if = { # can only do it once, except in de jure kingdom of current capital with the right ambition
				limit = { has_character_modifier = launched_subjugation }

				has_ambition = obj_become_king

				capital_scope = {
					kingdom = {
						title = PREVPREVPREV
					}
				}
			}
		}

		# Defender must have at least 1 county within the target kingdom
		num_title_realm_provs = {
			value >= 1
			who = FROM
		}
	}

	is_valid_title = {
		# Defender must have at least 1 county within the target kingdom
		num_title_realm_provs = {
			value >= 1
			who = FROM
		}
	}

	on_add = {
		attacker = {
			show_scope_change = no

			piety = -500

			if = {
				limit = {
					trigger_if = {
						limit = { has_ambition = obj_become_king }

						capital_scope = {
							kingdom = {
								NOT = { title = thirdparty_title_scope }
							}
						}
					}
				}

				add_character_modifier = {
					name = launched_subjugation
					years = 1000
					hidden = yes
				}
			}
		}

		anger_china_effect = yes
	}

	on_success = {
		any_attacker = {
			show_scope_change = no

			if = {
				limit = { character = attacker }

				if = {
					limit = { independent = no }
					liege = { save_event_target_as = old_liege }
				}

				# Usurp the target kingdom itself, if the enemy holds it
				if = {
					limit = {
						lower_real_tier_than = EMPEROR
						thirdparty_title_scope = { holder = defender }
					}

					usurp_title = {
						target = thirdparty_title_scope
						type = invasion
					}
				}

				if = {
					# Take the enemy's duchies within the target kingdom, if we are the same or lower tier
					limit = { NOT = { higher_real_tier_than = defender } }

					defender = {
						show_scope_change = no

						any_demesne_title = {
							show_scope_change = no

							limit = {
								real_tier = DUKE
								de_jure_liege_or_above = thirdparty_title_scope
							}

							usurp_title = {
								target = attacker
								type = invasion
							}
						}
					}
				}

				# Have I become the vassal of the loser due to taking a duchy or kingdom?
				if = {
					limit = { is_vassal_or_below_of = defender }

					if = {
						limit = { lower_real_tier_than = event_target:old_liege }
						set_defacto_liege = event_target:old_liege
					}
					else = {
						set_defacto_liege = THIS
					}
				}

				# Vassalize the rest
				custom_tooltip = {
					text = pagan_subjugation_tip

					subjugate_or_take_under_title = { # If the target only has territory within the kingdom, he is simply vassalized
						title = thirdparty_title_scope
						enemy = defender
					}
				}

				participation_scaled_prestige = 200
			}
			else = {
				hidden_effect = { participation_scaled_prestige = 200 }
			}
		}

		defender = {
			show_scope_change = no
			prestige = -200
		}
	}

	on_fail = {
		attacker = {
			show_scope_change = no
			prestige = -100
		}

		any_defender = {
			show_scope_change = no

			if = {
				limit = { character = defender }
				participation_scaled_prestige = 100
			}
			else = {
				hidden_effect = { participation_scaled_prestige = 100 }
			}
		}
	}

	on_reverse_demand = {
		attacker = {
			show_scope_change = no

			prestige = -200

			transfer_scaled_wealth = {
				to = defender
				value = 2.0
			}
		}

		any_defender = {
			show_scope_change = no

			if = {
				limit = { character = defender }
				participation_scaled_prestige = 200
			}
			else = {
				hidden_effect = { participation_scaled_prestige = 200 }
			}
		}
	}

	on_invalidation = {
		attacker = {
			show_scope_change = no

			remove_character_modifier = launched_subjugation
			piety = 500
		}
	}

	attacker_ai_victory_worth = {
		factor = -1 # always accept
	}

	attacker_ai_defeat_worth = {
		factor = 100
	}

	defender_ai_victory_worth = {
		factor = -1 # always accept
	}

	defender_ai_defeat_worth = {
		factor = 100
	}

	ai_will_do = {
		factor = 0.9 # Prefer using CB's without on_add costs

		trigger = {
			suitable_holdings_trigger = yes
			temujin_avoid_china_trigger = yes
		}
	}
}
