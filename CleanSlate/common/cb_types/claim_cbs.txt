claim = {
	name = CB_NAME_CLAIM
	war_name = WAR_NAME_CLAIM
	sprite = 1
	truce_days = 3650
	infamy_modifier = 0.85
	sort_priority = 1000

	hostile_against_others = yes
	can_ask_to_join_war = no
	allowed_to_target_tributaries = no
	press_claim = yes

	can_use_gui = {
		not_fellow_crusader_trigger = yes
	}

	can_use = {
		# Attacker
		ROOT = {
			# Gender requirements
			trigger_if = {
				limit = { is_female = yes }
				religion_is_patriarchal_trigger = no
			}
			trigger_else = {
				religion_is_matriarchal_trigger = no
			}
		}
	}

	can_use_title = {
		# Attacker
		ROOT = {
			# Claimant requirements
			OR = {
				has_strong_claim = PREV
				has_character_flag = claimant_adventurer

				trigger_if = {
					limit = { has_weak_claim = PREV }

					PREV = { # Claimed Title
						OR = {
							is_pretender = yes # Checks vs ROOT character
							is_contested = yes # Already being contested in some war
							holder_scope = { has_regent = yes }

							# A female ruler without full equality can be ousted by another female ruler with full equality
							trigger_if = {
								limit = {
									NOT = {
										has_game_rule = { # Makes the below moot
											name = gender
											value = all
										}
									}

									ROOT = { is_female = yes }

									holder_scope = {
										is_female = yes
										religion_is_matriarchal_trigger = no
										religion_is_equal_trigger = no
										NOT = { has_law = status_of_women_4 }
									}
								}

								ROOT = {
									OR = {
										has_law = status_of_women_4
										religion_is_matriarchal_trigger = yes
										religion_is_equal_trigger = yes
									}
								}
							}
						}
					}
				}
			}

			# Landless primary type rulers don't claim titles above their tier
			trigger_if = {
				limit = {
					OR = {
						mercenary = yes
						holy_order = yes

						trigger_if = {
							limit = { controls_religion = yes }
							is_theocracy = yes
						}
					}
				}

				NOT = { lower_real_tier_than = PREV }
			}
		}

		holder_scope = {
			OR = {
				character = FROM
				is_vassal_or_below_of = FROM
			}
		}
	}

	is_valid_title = {
		# Attacker
		claimed_by = ROOT

		holder_scope = {
			OR = {
				character = FROM
				is_vassal_or_below_of = FROM
			}
		}
	}

	on_add = {
		anger_china_effect = yes
	}

	on_success = {
		any_attacker = {
			show_scope_change = no

			if = {
				limit = { character = attacker }

				hidden_effect = {
					clr_character_flag = claimant_adventurer

					thirdparty_title_scope = {
						if = {
							limit = {
								higher_real_tier_than = BARON
								NOT = { culture = attacker }
							}

							conquest_culture = attacker
						}
					}

					if = {
						limit = {
							thirdparty_title_scope = {
								NOT = {
									any_previous_holder = { character = attacker }
								}
							}

							can_apply_average_nickname_trigger = yes
							lower_real_tier_than = thirdparty_title_scope
						}
					}

					random = {
						chance = 50
						give_nickname = nick_the_usurper
					}
				}

				usurp_title_plus_barony_if_unlanded = {
					target = thirdparty_title_scope
					type = claim
				}

				participation_scaled_prestige = 100
			}
			else = {
				hidden_effect = { participation_scaled_prestige = 100 }
			}
		}

		defender = {
			show_scope_change = no
			prestige = -100
		}
	}

	on_fail = {
		attacker = {
			show_scope_change = no

			prestige = -100

			hidden_effect = {
				disband_event_forces = yes
				clr_character_flag = claimant_adventurer
			}
		}

		any_defender = {
			show_scope_change = no

			if = {
				limit = { character = defender }
				participation_scaled_prestige = 100
			}
			else = {
				hidden_effect = { participation_scaled_prestige = 100 }
			}
		}
	}

	on_reverse_demand = {
		attacker = {
			show_scope_change = no

			remove_claim = thirdparty_title_scope
			prestige = -150

			transfer_scaled_wealth = {
				to = defender
				value = 4.0
			}

			if = {
				limit = { has_character_flag = claimant_adventurer }

				imprison = defender
				clr_character_flag = claimant_adventurer
			}

			hidden_effect = {
				disband_event_forces = yes

				# China stuff
				if = {
					limit = { has_landed_title = e_china_west_governor }

					log = "China War Logging: China lost a claim war against [From.GetBestName]"

					if = { # Makes China have a chance of leaving Expanionist policy...
						limit = {
							offmap_power = {
								has_policy = china_expansionist
							}
						}

						random = {
							chance = 50

							offmap_china = {
								set_policy = china_open
								log = "-------------------------------------"
								log = "China Policy Logging:"
								log = "China's new Policy is now OPEN"
								log = "-------------------------------------"
							}
						}
					}
				}
			}
		}

		any_defender = {
			show_scope_change = no

			if = {
				limit = { character = defender }
				participation_scaled_prestige = 150
			}
			else = {
				hidden_effect = { participation_scaled_prestige = 150 }
			}
		}
	}

	on_attacker_leader_death = {
		attacker = {
			show_scope_change = no

			if = {
				limit = {
					OR = {
						has_character_flag = claimant_adventurer
						has_character_flag = EoD_prophet_attacker
					}
				}

				hidden_effect = {
					if = {
						limit = { has_character_flag = EoD_prophet_attacker }

						defender = {
							narrative_event = { id = RIP.15116 }
						}
					}

					disband_event_forces = yes
					clr_character_flag = claimant_adventurer
					clr_character_flag = EoD_prophet_attacker
				}

				end_war = invalid
			}
		}
	}

	attacker_ai_victory_worth = {
		factor = -1 # always accept
	}

	attacker_ai_defeat_worth = {
		factor = 100
	}

	defender_ai_victory_worth = {
		factor = -1 # always accept
	}

	defender_ai_defeat_worth = {
		factor = 100
	}
}

claim_all = {
	name = CB_NAME_CLAIM_ALL

	sort_priority = 1100

	is_permanent = yes

	can_use_gui = {
		not_fellow_crusader_trigger = yes
	}

	can_use = {
		# Attacker
		ROOT = {
			total_claims = {
				who = FROM
				value >= 2
				strong = yes
			}
		}
	}

	is_valid = {
		# Attacker
		ROOT = {
			total_claims = {
				who = FROM
				value >= 1
				strong = yes
			}
		}
	}

	on_add = {
		attacker = {
			show_scope_change = no

			any_claim = {
				show_scope_change = no

				limit = {
					holder_scope = {
						OR = {
							character = defender
							is_vassal_or_below_of = defender
						}
					}

					attacker = { has_strong_claim = PREV }
				}

				press_claim = ROOT
			}
		}

		anger_china_effect = yes
	}

	on_success = {
		log = "[Root.GetBestName] (ID: [Root.GetID]) succeeded with [Root.GetHerHis] Claim All war against [From.GetBestName] (ID: [From.GetID])"

		any_attacker = {
			show_scope_change = no

			if = {
				limit = { character = attacker }

				participation_scaled_prestige = 100

				any_claim = {
					show_scope_change = no

					limit = {
						holder_scope = {
							OR = {
								character = defender
								is_vassal_or_below_of = defender
							}
						}

						attacker = { has_strong_claim = PREV }
					}

					hidden_effect = {
						if = {
							limit = {
								NOT = {
									any_previous_holder = { character = attacker }
								}

								attacker = {
									can_apply_average_nickname_trigger = yes
									lower_real_tier_than = PREV
								}
							}

							attacker = {
								random = {
									chance = 50
									give_nickname = nick_the_usurper
								}
							}
						}
					}

					usurp_title_plus_barony_if_unlanded = {
						target = attacker
						type = claim
					}

					hidden_effect = {
						if = {
							limit = {
								higher_real_tier_than = BARON
								NOT = { culture = attacker }
							}

							conquest_culture = attacker
						}
					}
				}

				hidden_effect = {
					if = { # China with no land calculates what to do with nomad provinces
						limit = {
							is_offmap_governor = yes
							offmap_power = { is_offmap_tag = offmap_china }
							offmap_china = { has_offmap_tmp_flag = WP_is_not_on_map }
							wealth < 2500 # unless already rich from tributes
						}

						set_variable = {
							which = local_WP_nomad_gold_from_china
							value = 0
						}

						any_realm_province = {
							limit = {
								has_province_modifier = nomad_agitation
								has_settlement_construction = no
								held_under_PREV = yes
							}

							change_variable = { which = local_WP_nomad_gold_from_china value = 1 }
						}

						character_event = {
							id = JD.60101 # calculate gold sent from china for building holdings
							days = 1
						}
					}

					if = {
						limit = {
							has_character_flag = is_otto

							defender = {
								has_character_flag = is_giselbert
							}
						}

						reverse_imprison = defender
					}
				}
			}
			else = {
				hidden_effect = { participation_scaled_prestige = 100 }
			}
		}

		defender = {
			show_scope_change = no
			prestige = -100
		}
	}

	on_fail = {
		attacker = {
			show_scope_change = no

			prestige = -150
		}

		any_defender = {
			show_scope_change = no

			if = {
				limit = { character = defender }
				participation_scaled_prestige = 75
			}
			else = {
				hidden_effect = { participation_scaled_prestige = 75 }
			}
		}
	}

	on_reverse_demand = {
		attacker = {
			show_scope_change = no

			prestige = -200

			transfer_scaled_wealth = {
				to = defender
				value = 4.0
			}

			any_claim = {

								log = "Chinas new Policy is now OPEN"
								log = "-------------------------------------"

		}

		any_defender = {

				is_vassal_or_below_of = FROM
			}
		}

		is_vice_royalty = no
	}

				target = thirdparty_title_scope
				type = claim
			}

			participation_scaled_prestige = 100

			# Attacker (ROOT) is freed from Defender (FROM)
			if = {
				limit = {
					prisoner = yes
					host = { character = defender }
				}

				imprison = no
			}

			hidden_effect = { disband_event_forces = faction_revolters }
		}

		any_attacker = {
			limit = { NOT = { character = defender } }

			hidden_effect = {
				participation_scaled_prestige = 100
				disband_event_forces = faction_revolters
			}
		}

		defender = {
			show_scope_change = no

			prestige = -100

			if = {
				limit = { primary_title = { title = thirdparty_title_scope } }

				imprison = attacker

				# Inform player about switch in feudal hierarchy

						limit = {
							NOT = {
								any_demesne_title = {

								}
							}
						}

						set_defacto_liege = attacker
					}
				}
			}

			hidden_effect = { disband_event_forces = faction_loyalists }

			# Lower crown authority
			any_demesne_title = {
				limit = { 
					is_crown_law_title = yes 
					is_law_potential = crown_authority_0
				}

				trigger_switch = {
					on_trigger = has_law

					crown_authority_1 = { add_law = crown_authority_0 }
					crown_authority_2 = { add_law = crown_authority_1 }
					crown_authority_3 = { add_law = crown_authority_2 }
					crown_authority_4 = { add_law = crown_authority_3 }
				}
			}
		}

		hidden_effect = {
			thirdparty_title_scope = {
				# Conquest Culture
				if = {
					limit = {
						higher_real_tier_than = BARON
						NOT = { culture = attacker }
					}

					conquest_culture = attacker
				}
			}
		}
	}

	on_fail = {
		attacker = {
			show_scope_change = no

			prestige = -100
			clr_character_flag = faction_claimant_ultimatum_taken_@thirdparty_title_scope
			hidden_effect = { disband_event_forces = faction_revolters }
		}

		any_attacker = {
			limit = { vassal_of = defender }
			hidden_effect = { disband_event_forces = faction_revolters }
		}

		defender = {
			show_scope_change = no

			participation_scaled_prestige = 50
			hidden_effect = { disband_event_forces = faction_loyalists }
		}

		any_defender = {
			limit = { NOT = { character = defender } }
			hidden_effect = { participation_scaled_prestige = 50 }
		}
	}

	on_reverse_demand = {
		attacker = {
			show_scope_change = no

			prestige = -150
			clr_character_flag = faction_claimant_ultimatum_taken_@thirdparty_title_scope

								}
							}
						}

							name = opinion_traitor
							who = attacker
						}
					}

			}
			else = {

					participation_scaled_prestige = 100
					disband_event_forces = faction_loyalists
				}
			}
		}
	}

		}

				}
				else = {
					set_defacto_liege = THIS
				}
			}

			# Vassalize to attacker
			else_if = {
				limit = {
					lower_real_tier_than = event_target:attacker

					OR = {
						thirdparty_title_scope = {
							de_jure_liege_or_above = event_target:attacker
						}

					target = thirdparty_title_scope
					type = claim

				}

				limit = {

					}

					type = claim
				}

			}

					value = all
				}

					name = pressed_my_claim

				}
			}

		}

			show_scope_change = no

			hidden_effect = {
				disband_event_forces = faction_revolters

				if = {
					limit = {
						thirdparty_title_scope = {
							NOT = {
								any_previous_holder = { character = event_target:claimant }
							}
						}

						can_apply_average_nickname_trigger = yes
						lower_real_tier_than = thirdparty_title_scope
					}

					random = {
						chance = 50
						give_nickname = nick_the_usurper
					}
				}

				# If the claimant has the wrong government type, take care of that first.
				if = {
					limit = {
						is_ruler = yes

						OR = {
							is_theocracy = yes
							is_republic = yes
						}

						thirdparty_title_scope = {
							higher_real_tier_than = event_target:claimant
							holder_scope = { is_feudal = yes }
						}
					}

					abdicate = { move = no }
				}
			}

			participation_scaled_prestige = 100

			if = {
				limit = {
					prisoner = yes
					host = { character = defender }
				}

				imprison = no
			}

			usurp_title_plus_barony_if_unlanded_and_vassals_no_adj = {
				target = thirdparty_title_scope
				type = claim
			}

			set_defacto_liege = event_target:new_liege

			# Lower crown authority
			thirdparty_title_scope = {
				if = {
					limit = {
						is_crown_law_title = yes
						is_law_potential = crown_authority_0
					}

					trigger_switch = {
						on_trigger = has_law

						crown_authority_1 = { add_law = crown_authority_0 }
						crown_authority_2 = { add_law = crown_authority_1 }
						crown_authority_3 = { add_law = crown_authority_2 }
						crown_authority_4 = { add_law = crown_authority_3 }
					}
				}
			}

			opinion = {
				name = pressed_my_claim
				who = event_target:attacker
			}
		}

		any_attacker = {
			limit = {
				NOR = {
					character = event_target:claimant
					character = event_target:attacker
				}
			}

			hidden_effect = {
				participation_scaled_prestige = 100
				disband_event_forces = faction_revolters
			}
		}

		# Defender
		defender = {
			show_scope_change = no

			prestige = -100
			hidden_effect = { disband_event_forces = faction_loyalists }

			if = {
				limit = {
					# For tooltip, since no title transfer has happened yet
					trigger_if = {
						limit = { has_landed_title = thirdparty_title_scope }

						# Will be a subject of claimant
						NOT = {
							any_demesne_title = {
								NOT = {
									title = thirdparty_title_scope
									lower_real_tier_than = thirdparty_title_scope
								}
							}
						}
					}
					# The actual check we need
					trigger_else = {
						lower_real_tier_than = event_target:claimant
					}
				}

				imprison = event_target:claimant
			}

			# Lower crown authority
			any_demesne_title = {
				limit = {
					is_crown_law_title = yes
					is_law_potential = crown_authority_0
					NOT = { title = thirdparty_title_scope } # For the tooltip, this title changes hands
				}

				trigger_switch = {
					on_trigger = has_law

					crown_authority_1 = { add_law = crown_authority_0 }
					crown_authority_2 = { add_law = crown_authority_1 }
					crown_authority_3 = { add_law = crown_authority_2 }
					crown_authority_4 = { add_law = crown_authority_3 }
				}
			}
		}

		hidden_effect = {
			thirdparty_title_scope = {
				if = {
					limit = {
						higher_real_tier_than = BARON
						NOT = { culture = event_target:claimant }
					}

					conquest_culture = event_target:claimant
				}
			}
		}
	}

	# Starting scope is attacker
	# ROOT is claimant
	# FROM is defender
	# FROMFROM is title
	on_fail = {
		# Attacker
		prestige = -100
		imprison = defender
		clr_character_flag = faction_claimant_ultimatum_taken_@PREV
		disband_event_forces = faction_revolters
		ROOT = { save_event_target_as = claimant }

		event_target:claimant = {
			show_scope_change = no
			prestige = -100
		}

		any_attacker = {
			show_scope_change = no

			hidden_effect = {
				disband_event_forces = faction_revolters
			}
		}

		any_defender = {
			if = {
				limit = { character = defender }

				participation_scaled_prestige = 50
				hidden_effect = { disband_event_forces = faction_loyalists }
			}
			else = {
				hidden_effect = { participation_scaled_prestige = 50 }
			}
		}
	}

	# Starting scope is attacker
	# ROOT is claimant
	# FROM is defender
	# FROMFROM is title
	on_reverse_demand = {
		# Atacker
		prestige = -200
		imprison = defender
		clr_character_flag = faction_claimant_ultimatum_taken_@thirdparty_title_scope
		ROOT = { save_event_target_as = claimant }

		event_target:claimant = {
			show_scope_change = no
			prestige = -200
		}

		hidden_effect = {
			any_attacker = {
				limit = { vassal_of = defender }

				disband_event_forces = faction_revolters
				imprison = defender
			}
		}

		any_defender = {
			show_scope_change = no

			if = {
				limit = { character = defender }

				participation_scaled_prestige = 100

				hidden_effect = {
					disband_event_forces = faction_loyalists

					if = { # Allow free revocation of a title
						limit = {
							NOT = {
								has_opinion_modifier = {
									name = opinion_traitor
									who = event_target:claimant
								}
							}
						}

						opinion = {
							name = opinion_traitor
							who = event_target:claimant
						}
					}
				}
			}
			else = {
				hidden_effect = { participation_scaled_prestige = 100 }
			}
		}
	}

	on_invalidation = {
		clr_character_flag = faction_claimant_ultimatum_taken_@thirdparty_title_scope
	}

	attacker_ai_victory_worth = {
		factor = -1 # always accept
	}

	attacker_ai_defeat_worth = {
		factor = 100
	}

	defender_ai_victory_worth = {
		factor = -1 # always accept
	}

	defender_ai_defeat_worth = {
		factor = 100
	}
